    function replaceTriggerWord(newText) {
        const val = nlInput.value;
        const cursor = nlInput.selectionStart;
        const textBefore = val.substring(0, cursor);
        const lastSlash = textBefore.lastIndexOf('/');
        const lastAt = textBefore.lastIndexOf('@');
        const lastTilde = textBefore.lastIndexOf('~');
        const triggerPos = Math.max(lastSlash, lastAt, lastTilde);

        if (triggerPos === -1) {
            nlInput.value += newText; 
        } else {
            const textAfter = val.substring(cursor);
            if(val[triggerPos] === '/' && triggerPos === 0) {
                 nlInput.value = newText + " ";
            } else {
                 const beforeTrigger = val.substring(0, triggerPos);
                 nlInput.value = beforeTrigger + newText + textAfter;
            }
        }
        nlInput.focus();
    }

    function insertTag(text) {
        const val = nlInput.value;
        if (val.match(/^\d{4}-\d{2}-\d{2},/)) nlInput.value = val.replace(/^\d{4}-\d{2}-\d{2},/, text);
        else if (val.toLowerCase().startsWith('today,')) nlInput.value = val.replace(/^today,\s*/i, text);
        else if (val.toLowerCase().startsWith('yesterday,')) nlInput.value = val.replace(/^yesterday,\s*/i, text);
        else replaceTriggerWord(text); 
        hideMenus(); nlInput.focus();
    }

    // --- PARSING & HANDLERS ---
    function handleInput() {
        hideMenus();
        const text = nlInput.value.trim();
        if(!text) return showToast('Input kosong', true);

        if(text.startsWith('/')) {
            handleCommand(text);
        } else {
            const res = parseNL(text);
            if(!res) return showToast('Format salah', true);
            if(res.type === 'Expense' && isIncomeCategory(res.category)) showWarningModal(res);
            else showPreview(res);
        }
    }

    function isIncomeCategory(cat) {
        if(!cat) return false;
        const lower = cat.toLowerCase();
        return INCOME_CATEGORIES.some(inc => lower.includes(inc) || (CATEGORY_MAP_CLIENT[inc] && CATEGORY_MAP_CLIENT[inc].toLowerCase() === lower));
    }

    function parseNL(text) {
        // 1. Expand keywords (Moved Date extraction logic here to ensure order)
        let clean = text.replace(/today/i, new Date().toISOString().split('T')[0])
                        .replace(/yesterday/i, new Date(Date.now() - 864e5).toISOString().split('T')[0])
                        .replace(/~/, '');
        
        // 2. Handle explicit category shortcuts
        const atMatch = clean.match(/@([\w-]+)/);
        if(atMatch && atMatch[1] && CATEGORY_MAP_CLIENT[atMatch[1].toLowerCase()]) 
            clean = clean.replace(atMatch[0], CATEGORY_MAP_CLIENT[atMatch[1].toLowerCase()]);
        
        // 3. EXTRACT DATE FIRST (Fix for date getting corrupted by '-' check)
        let date = new Date();
        // Regex to capture date at the beginning of string
        const dateMatch = clean.match(/^(\d{4}-\d{2}-\d{2})/);
        if(dateMatch) {
            date = new Date(dateMatch[1]);
            // Remove the date string
            clean = clean.replace(dateMatch[0], '');
            // Clean up any remaining leading comma/space
            clean = clean.replace(/^[\s,]+/, '').trim();
        }

        // 4. Determine Type (+/-) AFTER date extraction
        let type = 'Expense';
        if (clean.includes('+')) { type = 'Income'; clean = clean.replace('+', ''); }
        else if (clean.includes('-')) { type = 'Expense'; clean = clean.replace('-', ''); }

        // 5. Parse Amount, Category, Notes
        const parts = clean.split(/[\s,]+/);
        let amount = 0, note = [];
        parts.forEach(p => {
            const num = p.replace(/\./g, '').replace(',', '.');
            if(!isNaN(num) && num !== '' && amount === 0) amount = parseFloat(num);
            else if(p.length > 0) note.push(p);
        });

        if(amount === 0) return null;
        let cat = 'Others';
        let finalNote = [];
        note.forEach(w => {
            const mapped = CATEGORY_MAP_CLIENT[w.toLowerCase()];
            if(mapped) cat = mapped; 
            else finalNote.push(w);
        });

        return { date: date, type: type, amount: amount, category: cat, note: finalNote.join(' ') || cat };
    }
